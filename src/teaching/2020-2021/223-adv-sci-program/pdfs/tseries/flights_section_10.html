feb = time_index.loc['2013-02'][['dep_delay', 'temp', 'precip', 'wind_speed']].resample('D').agg({
    'dep_delay': 'count',
    'temp': 'mean',
    'precip': 'sum',
    'wind_speed': 'mean'
})

ax = feb.plot(kind='bar', subplots=True, layout=(4,1), legend=False, figsize=(15, 15))
